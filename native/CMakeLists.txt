cmake_minimum_required(VERSION 3.18)
project(pyrpmd_native LANGUAGES CXX)
find_package(Python3 REQUIRED COMPONENTS Interpreter Development.Module)
find_package(OpenMP)
Python3_add_library(_native MODULE WITH_SOABI cayley.cpp)
target_compile_features(_native PRIVATE cxx_std_17)
if(OpenMP_CXX_FOUND)
    target_link_libraries(_native PRIVATE OpenMP::OpenMP_CXX)
endif()
install(TARGETS _native LIBRARY DESTINATION pyrpmd)
install(FILES include/pyrpmd/pes_api_v1.h DESTINATION include/pyrpmd)
