cmake_minimum_required(VERSION 3.18)
project(pyrpmd_native LANGUAGES CXX)
find_package(Python3 REQUIRED COMPONENTS Interpreter Development.Module)
find_package(OpenMP)
Python3_add_library(_native MODULE WITH_SOABI cayley.cpp)
Python3_add_library(_thermal MODULE WITH_SOABI thermal.cpp)
foreach(target _native _thermal)
    target_compile_features(${target} PRIVATE cxx_std_17)
    if(OpenMP_CXX_FOUND)
        target_link_libraries(${target} PRIVATE OpenMP::OpenMP_CXX)
    endif()
endforeach()
install(TARGETS _native _thermal LIBRARY DESTINATION pyrpmd)
install(FILES include/pyrpmd/pes_api_v1.h include/pyrpmd/pes_api_v2.h include/pyrpmd/thermal_pes.h DESTINATION include/pyrpmd)
