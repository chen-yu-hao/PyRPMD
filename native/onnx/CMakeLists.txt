cmake_minimum_required(VERSION 3.18)
project(pyrpmd_onnx LANGUAGES CXX)
# Optional standalone plugin; the ordinary PyRPMD build never requires ORT.
set(ONNXRUNTIME_ROOT "" CACHE PATH "Unpacked ONNX Runtime C/C++ SDK")
set(PYRPMD_ONNX_CUDA "AUTO" CACHE STRING "Build CUDA device-memory evaluation support (AUTO, ON, OFF)")
set_property(CACHE PYRPMD_ONNX_CUDA PROPERTY STRINGS AUTO ON OFF)

find_path(ONNXRUNTIME_INCLUDE_DIR onnxruntime_cxx_api.h
  HINTS "${ONNXRUNTIME_ROOT}" ENV ONNXRUNTIME_ROOT
  PATH_SUFFIXES include include/onnxruntime/core/session REQUIRED)
find_library(ONNXRUNTIME_LIBRARY NAMES onnxruntime
  HINTS "${ONNXRUNTIME_ROOT}" ENV ONNXRUNTIME_ROOT PATH_SUFFIXES lib lib64 REQUIRED)
find_package(nlohmann_json 3.10 REQUIRED)

set(_pyrpmd_onnx_enable_cuda FALSE)
if(PYRPMD_ONNX_CUDA STREQUAL "ON")
  find_package(CUDAToolkit REQUIRED)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    if(CUDAToolkit_VERSION VERSION_GREATER_EQUAL "13.0")
      set(CMAKE_CUDA_ARCHITECTURES "75;80;86;89;90")
    elseif(CUDAToolkit_VERSION VERSION_GREATER_EQUAL "11.8")
      set(CMAKE_CUDA_ARCHITECTURES "70;75;80;86;89;90")
    else()
      set(CMAKE_CUDA_ARCHITECTURES "70;75;80;86")
    endif()
  endif()
  enable_language(CUDA)
  set(_pyrpmd_onnx_enable_cuda TRUE)
elseif(PYRPMD_ONNX_CUDA STREQUAL "AUTO")
  include(CheckLanguage)
  check_language(CUDA)
  find_package(CUDAToolkit QUIET)
  if(CMAKE_CUDA_COMPILER AND CUDAToolkit_FOUND)
    if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
      if(CUDAToolkit_VERSION VERSION_GREATER_EQUAL "13.0")
        set(CMAKE_CUDA_ARCHITECTURES "75;80;86;89;90")
      elseif(CUDAToolkit_VERSION VERSION_GREATER_EQUAL "11.8")
        set(CMAKE_CUDA_ARCHITECTURES "70;75;80;86;89;90")
      else()
        set(CMAKE_CUDA_ARCHITECTURES "70;75;80;86")
      endif()
    endif()
    enable_language(CUDA)
    set(_pyrpmd_onnx_enable_cuda TRUE)
  endif()
elseif(NOT PYRPMD_ONNX_CUDA STREQUAL "OFF")
  message(FATAL_ERROR "PYRPMD_ONNX_CUDA must be AUTO, ON, or OFF")
endif()

set(_pyrpmd_onnx_sources onnx_pes.cpp)
if(_pyrpmd_onnx_enable_cuda)
  list(APPEND _pyrpmd_onnx_sources onnx_cuda.cu)
  message(STATUS "pyrpmd_onnx: enabling CUDA device-memory support (CUDAToolkit ${CUDAToolkit_VERSION})")
else()
  message(STATUS "pyrpmd_onnx: building host-memory only (PYRPMD_ONNX_CUDA=${PYRPMD_ONNX_CUDA})")
endif()

add_library(pyrpmd_onnx SHARED ${_pyrpmd_onnx_sources})
target_compile_features(pyrpmd_onnx PRIVATE cxx_std_17)
target_include_directories(pyrpmd_onnx PRIVATE ../include "${ONNXRUNTIME_INCLUDE_DIR}")
target_link_libraries(pyrpmd_onnx PRIVATE "${ONNXRUNTIME_LIBRARY}" nlohmann_json::nlohmann_json)
set_target_properties(pyrpmd_onnx PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON POSITION_INDEPENDENT_CODE ON)

if(_pyrpmd_onnx_enable_cuda)
  target_compile_definitions(pyrpmd_onnx PRIVATE PYRPMD_ONNX_HAS_CUDA=1)
  set_target_properties(pyrpmd_onnx PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  if(TARGET CUDA::cudart_static)
    target_link_libraries(pyrpmd_onnx PRIVATE CUDA::cudart_static)
  else()
    target_link_libraries(pyrpmd_onnx PRIVATE CUDA::cudart)
  endif()
else()
  target_compile_definitions(pyrpmd_onnx PRIVATE PYRPMD_ONNX_HAS_CUDA=0)
endif()

if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(pyrpmd_onnx PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall -Wextra -Wpedantic>)
endif()

# Test a native multi-context caller, independently of Python's conservative
# per-library lock. This does NOT replace PyRPMD's trajectory scheduler.
option(PYRPMD_ONNX_CONTEXT_TEST "Build the OpenMP context-isolation test" OFF)
if(PYRPMD_ONNX_CONTEXT_TEST)
  find_package(OpenMP REQUIRED)
  add_executable(onnx_context_test context_test.cpp)
  target_compile_features(onnx_context_test PRIVATE cxx_std_17)
  target_include_directories(onnx_context_test PRIVATE ../include)
  target_link_libraries(onnx_context_test PRIVATE pyrpmd_onnx OpenMP::OpenMP_CXX nlohmann_json::nlohmann_json)
endif()
install(TARGETS pyrpmd_onnx LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
